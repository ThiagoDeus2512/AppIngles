        const LIB = {
            pronomes: ['i', 'you', 'he', 'she', 'it', 'we', 'they', 'my', 'your', 'his', 'her', 'this', 'that'],
            auxiliares: ['do', 'does', 'did', 'will', 'would', 'can', 'could', 'should', 'must', 'am', 'is', 'are', 'was', 'were', 'have', 'has', 'had'],
            verbos: ['go', 'want', 'like', 'play', 'work', 'see', 'take', 'get', 'know', 'speak', 'study', 'believe', 'understand'],
            adjetivos: ['good', 'bad', 'happy', 'sad', 'beautiful', 'big', 'small', 'hot', 'cold', 'new', 'old', 'easy', 'hard', 'important', 'ready']
        };

        // --- INÍCIO DA CAMADA DE PRONÚNCIA DINÂMICA ---
        const REGRAS_PRONUNCIA = [
            { r: /\bdid you\b/gi, s: "did-ju" },
            { r: /\bwant to\b/gi, s: "wanna" },
            { r: /\bgoing to\b/gi, s: "gonna" },
            { r: /\bgot to\b/gi, s: "gotta" },
            { r: /\bi would\b/gi, s: "aid" },
            { r: /\bwhat do you\b/gi, s: "wha-du-ya" }
        ];

        function gerarPronunciaDinamica(texto, originalSound) {
            let temp = texto.toLowerCase();
            let temConexao = false;
            REGRAS_PRONUNCIA.forEach(regra => {
                if (regra.r.test(temp)) {
                    temp = temp.replace(regra.r, `<span class="connected-match">${regra.s}</span>`);
                    temConexao = true;
                }
            });
            return temConexao ? `Conexão: ${temp}` : `Som: ${originalSound}`;
        }
        // --- FIM DA CAMADA DE PRONÚNCIA ---

        function applySmartChunking(text) {
            const commonChunks = [
                /\b(i am|i'm|it is|it's|you are|you're|they are|they're|we are|we're)\b/gi,
                /\b(want to|wanna|going to|gonna|have to|has to|need to|did you|do you|will you|would you)\b/gi,
                /\b(i don't|i didn't|it doesn't|she doesn't)\b/gi
            ];
            let processed = text;
            commonChunks.forEach(regex => { processed = processed.replace(regex, (match) => match.replace(/ /g, "_")); });
            let words = processed.split(' ');
            let finalBlocks = [];
            for (let i = 0; i < words.length; i++) {
                if (words[i].includes('_')) {
                    finalBlocks.push(words[i].replace(/_/g, " "));
                } else if (nivelAtivo === 'ADVANCED') {
                    if (words[i + 1] && words[i + 2]) { finalBlocks.push(`${words[i]} ${words[i + 1]} ${words[i + 2]}`); i += 2; }
                    else { finalBlocks.push(words[i]); }
                } else if (words[i + 1] && !words[i + 1].includes('_')) {
                    finalBlocks.push(words[i] + " " + words[i + 1]); i++;
                } else { finalBlocks.push(words[i]); }
            }
            return finalBlocks;
        }

        function handleTyping() {
            const input = document.getElementById('userInput');
            const currentItem = getCurrentItem();
            if (!currentItem) return;
            const target = currentItem.pt.toLowerCase().trim();
            if (target.startsWith(input.value.toLowerCase().trim())) {
                input.classList.add('correct-path');
            } else {
                input.classList.remove('correct-path');
            }
            updateGhostHint();
        }

        function updateGhostHint() {
            const input = document.getElementById('userInput');
            const ghost = document.getElementById('ghostHint');
            const itemAtual = getCurrentItem();
            if (!itemAtual || input.value.length > 0) {
                ghost.style.display = 'none';
                return;
            }
            ghost.innerText = itemAtual.pt.charAt(0) + "...";
            ghost.style.display = 'block';
        }

        function getCurrentItem() {
            const filtradas = frases.filter(f => f.meta.level === nivelAtivo);
            const display = document.getElementById('blockDisplay');
            if (!display) return null;
            const textoNoBloco = display.innerText.toLowerCase().replace(/\n/g, " ");
            return filtradas.find(f => f.en.toLowerCase() === textoNoBloco) || filaSRS.find(f => f.en.toLowerCase() === textoNoBloco);
        }

        function getEstiloDinamico(texto) {
            const tokens = texto.toLowerCase().split(' ');
            if (tokens.some(t => LIB.auxiliares.includes(t))) return 'aux-glow';
            if (tokens.some(t => LIB.verbos.includes(t))) return 'verb-glow';
            return '';
        }

        function getCorGramatical(texto) {
            const tokens = texto.toLowerCase().split(' ');
            if (tokens.some(t => LIB.auxiliares.includes(t))) return 'border-bottom: 4px solid var(--auxiliary);';
            if (tokens.some(t => LIB.pronomes.includes(t))) return 'border-bottom: 4px solid var(--pronoun);';
            if (tokens.some(t => LIB.verbos.includes(t))) return 'border-bottom: 4px solid var(--verb);';
            if (tokens.some(t => LIB.adjetivos.includes(t))) return 'border-bottom: 4px solid var(--adjective);';
            return 'border-bottom: 4px solid #444;';
        }

        function showXpBonus(amount) {
            const div = document.createElement('div');
            div.className = 'xp-float';
            div.innerText = `+${amount} XP`;
            document.body.appendChild(div);
            setTimeout(() => div.remove(), 1000);
        }

        function playFeedback(type) {
            const context = new (window.AudioContext || window.webkitAudioContext)();
            const osc = context.createOscillator();
            const gain = context.createGain();
            osc.connect(gain); gain.connect(context.destination);
            if (type === 'success') {
                osc.type = 'sine'; osc.frequency.setValueAtTime(587, context.currentTime);
                osc.frequency.exponentialRampToValueAtTime(880, context.currentTime + 0.1);
            } else { osc.type = 'square'; osc.frequency.setValueAtTime(110, context.currentTime); }
            gain.gain.setValueAtTime(0.05, context.currentTime);
            osc.start(); osc.stop(context.currentTime + 0.15);
        }

        function triggerErrorEffect() {
            const container = document.getElementById('mainContainer');
            container.classList.add('shake-error');
            playFeedback('error');
            setTimeout(() => container.classList.remove('shake-error'), 300);
        }

        async function fetchFrases() {
            try {
                const res = await fetch(API_URL);
                frases = await res.json();
                atualizarEstatisticas();
                if (filaSRS.length > 0) document.getElementById('reviewPanel').style.display = 'block';
                render();
            } catch (e) { console.error("Servidor Offline"); render(); }
        }

        function atualizarEstatisticas() {
            let p = 0, v = 0, a = 0;
            frases.forEach(f => {
                const words = f.en.toLowerCase().split(' ');
                words.forEach(word => {
                    if (LIB.pronomes.includes(word)) p++;
                    if (LIB.verbos.includes(word) || LIB.auxiliares.includes(word)) v++;
                    if (LIB.adjetivos.includes(word)) a++;
                });
            });
            document.getElementById('count-pronouns').innerText = p;
            document.getElementById('count-verbs').innerText = v;
            document.getElementById('count-adjectives').innerText = a;
        }

        async function analisarDinamico() {
            const en = document.getElementById('newEn').value;
            const preview = document.getElementById('translationPreview');
            if (!en) { resetTags(); preview.innerText = "Aguardando entrada..."; return; }
            let tense = "PRESENT";
            if (/\b(will|going to|'ll)\b/i.test(en)) tense = "FUTURE";
            else if (/\b(ed|did|was|were|went|had|could|would)\b/i.test(en)) tense = "PAST";
            let isQuestion = en.includes("?");
            let isNegative = /\b(not|n't|never|no)\b/i.test(en);
            resetTags();
            document.getElementById(`tag-${tense}`).classList.add('active');
            if (isQuestion) document.getElementById('tag-QUESTION').classList.add('active');
            if (isNegative) document.getElementById('tag-NEGATIVE').classList.add('active');

            clearTimeout(timeoutTraducao);
            timeoutTraducao = setTimeout(async () => {
                try {
                    const res = await fetch(`https://api.mymemory.translated.net/get?q=${encodeURIComponent(en)}&langpair=en|pt-BR`);
                    const data = await res.json();
                    document.getElementById('newPt').value = data.responseData.translatedText;
                    preview.innerText = "✨ Tradução sugerida pronta!";
                } catch (e) { preview.innerText = "Erro na tradução"; }
            }, 600);
            analiseAtual = { tense, type: isQuestion ? "QUESTION" : (isNegative ? "NEGATIVE" : "AFFIRMATIVE"), blocks: applySmartChunking(en) };
        }

        function resetTags() { document.querySelectorAll('.ai-tag').forEach(t => t.classList.remove('active')); }

        async function saveNewPhrase() {
            const en = document.getElementById('newEn').value;
            const pt = document.getElementById('newPt').value;
            if (!en || !pt) return alert("Preencha o Inglês e a Tradução!");
            const novaFrase = {
                en, pt, blocks: analiseAtual.blocks,
                meta: { level: document.getElementById('newLevel').value, tense: analiseAtual.tense, type: analiseAtual.type, soundsLike: document.getElementById('manualSounds').value || en }
            };
            await fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(novaFrase) });
            location.reload();
        }

        function render() {
            const filtradas = frases.filter(f => f.meta.level === nivelAtivo);
            document.getElementById('xpDisplay').innerText = xp;
            const perc = filtradas.length > 0 ? (indexAtual / filtradas.length) * 100 : 0;
            document.getElementById('progressBar').style.width = perc + "%";

            let item;
            if (filaSRS.length > 0 && Math.random() > 0.6) {
                item = filaSRS[Math.floor(Math.random() * filaSRS.length)];
            } else {
                if (indexAtual >= filtradas.length && filtradas.length > 0) {
                    document.getElementById('blockDisplay').innerHTML = `<h3>Missão Cumprida! 🚀</h3>`;
                    clearInterval(timerFunc); return;
                }
                item = filtradas[indexAtual];
            }

            if (!item) return;
            document.getElementById('tenseBadge').innerText = item.meta.tense;
            document.getElementById('tenseBadge').className = `badge tense-${item.meta.tense.toLowerCase()}`;
            document.getElementById('typeBadge').innerText = item.meta.type;

            // ATUALIZAÇÃO DA LINHA DE PRONÚNCIA (Usando a nova função dinâmica)
            document.getElementById('pronunciationDisplay').innerHTML = gerarPronunciaDinamica(item.en, item.meta.soundsLike);

            const display = document.getElementById('blockDisplay');
            display.innerHTML = '';
            const blocosDinamicos = applySmartChunking(item.en);
            blocosDinamicos.forEach(text => {
                const div = document.createElement('div');
                div.className = `chunk ${getEstiloDinamico(text)}`;
                div.innerText = text;
                div.style = getCorGramatical(text);
                display.appendChild(div);
            });
            falar(item.en);
            startTimer();
            updateGhostHint();
            renderReviewList();
        }

        function renderReviewList() {
            const list = document.getElementById('reviewList');
            list.innerHTML = filaSRS.slice(0, 5).map(f => {
                let dots = '';
                for (let i = 0; i < 3; i++) dots += `<div class="dot ${i < (f.mastery || 0) ? 'filled' : ''}"></div>`;
                return `
                <div style="padding:10px; font-size: 0.85rem; border-bottom:1px solid #222; display: flex; justify-content: space-between; align-items: center;">
                    <b style="color: #ccc">${f.en}</b>
                    <div class="mastery-dots">${dots}</div>
                </div>`;
            }).join('');
        }

        function startTimer() {
            clearInterval(timerFunc); timeLeft = 15;
            document.getElementById('timerDisplay').innerText = timeLeft;
            timerFunc = setInterval(() => {
                timeLeft--; document.getElementById('timerDisplay').innerText = timeLeft;
                if (timeLeft <= 0) { clearInterval(timerFunc); triggerErrorEffect(); registrarErro(); proximaFrase(); }
            }, 1000);
        }

        function registrarErro() {
            const itemAtual = getCurrentItem();
            if (!itemAtual) return;
            const index = filaSRS.findIndex(x => x.en === itemAtual.en);
            if (index === -1) {
                filaSRS.push({ ...itemAtual, mastery: 0, lastReview: Date.now() });
            } else {
                filaSRS[index].mastery = 0;
                filaSRS[index].lastReview = Date.now();
            }
            saveSRS();
            document.getElementById('reviewPanel').style.display = 'block';
        }

        function saveSRS() { localStorage.setItem('dev_srs_queue_v2', JSON.stringify(filaSRS)); }

        function handleEnter(e) {
            if (e.key === 'Enter') {
                const input = document.getElementById('userInput').value.trim().toLowerCase().replace(/[?.,!]/g, "");
                const itemAtual = getCurrentItem();
                if (!itemAtual) return;
                const correct = itemAtual.pt.toLowerCase().replace(/[?.,!]/g, "");
                if (input === correct) {
                    playFeedback('success');
                    const speedBonus = timeLeft * 2;
                    xp += (10 + speedBonus);
                    showXpBonus(10 + speedBonus);
                    localStorage.setItem('dev_xp', xp);
                    const srsIndex = filaSRS.findIndex(f => f.en === itemAtual.en);
                    if (srsIndex !== -1) {
                        filaSRS[srsIndex].mastery = (filaSRS[srsIndex].mastery || 0) + 1;
                        filaSRS[srsIndex].lastReview = Date.now();
                        if (filaSRS[srsIndex].mastery >= 3) { filaSRS.splice(srsIndex, 1); }
                    }
                    saveSRS();
                    if (filaSRS.length === 0) document.getElementById('reviewPanel').style.display = 'none';
                    proximaFrase();
                } else { triggerErrorEffect(); registrarErro(); }
            }
        }

        function proximaFrase() { indexAtual++; document.getElementById('userInput').value = ''; document.getElementById('heardText').innerText = ''; render(); }

        function falar(t) {
            window.speechSynthesis.cancel();
            const u = new SpeechSynthesisUtterance(t);
            u.lang = 'en-US'; u.rate = 0.85;
            window.speechSynthesis.speak(u);
        }

        function falarAtual() { const display = document.getElementById('blockDisplay'); if (display) falar(display.innerText); }

        function setNivel(lvl) {
            nivelAtivo = lvl; indexAtual = 0;
            document.querySelectorAll('.level-btn').forEach(b => b.classList.remove('active'));
            document.getElementById(`btn-${lvl}`).classList.add('active');
            render();
        }

        function testarPronuncia() {
            const Speech = window.SpeechRecognition || window.webkitSpeechRecognition;
            if (!Speech) return;
            const rec = new Speech(); rec.lang = 'en-US';
            rec.onstart = () => document.getElementById('wave').classList.add('active');
            rec.onend = () => document.getElementById('wave').classList.remove('active');
            rec.onresult = (e) => {
                const talk = e.results[0][0].transcript.toLowerCase().replace(/[?.,!]/g, "");
                const textoNoBloco = document.getElementById('blockDisplay').innerText.toLowerCase().replace(/\n/g, " ");
                if (talk.includes(textoNoBloco) || textoNoBloco.includes(talk)) {
                    playFeedback('success');
                    const speedBonus = timeLeft * 5;
                    xp += (50 + speedBonus); showXpBonus(50 + speedBonus);
                    proximaFrase();
                } else {
                    document.getElementById('heardText').innerText = `Ouvi: "${talk}"`;
                    triggerErrorEffect();
                }
            };
            rec.start();
        }

        window.onload = fetchFrases;
